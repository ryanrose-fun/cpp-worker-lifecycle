cmake_minimum_required(VERSION 3.20)
project(worker_lifecycle LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(worker INTERFACE)
target_include_directories(worker INTERFACE include)
target_compile_features(worker INTERFACE cxx_std_20)
target_link_libraries(worker INTERFACE Threads::Threads)
if(MSVC)
  target_compile_options(worker INTERFACE /W4 /WX /permissive-)
else()
  target_compile_options(worker INTERFACE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(worker_demo src/main.cpp)
target_link_libraries(worker_demo PRIVATE worker)
enable_testing()
add_executable(worker_tests tests/worker_tests.cpp)
target_link_libraries(worker_tests PRIVATE worker)
add_test(NAME lifecycle COMMAND worker_tests)
set_tests_properties(lifecycle PROPERTIES TIMEOUT 30)
